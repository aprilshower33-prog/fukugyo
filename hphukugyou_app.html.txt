function normalizeLoadedData(data){ return (data && typeof data==='object') ? data : null; }
function loadFromLocalStorage(){
  try{
    const raw=localStorage.getItem(STORAGE_KEY)||localStorage.getItem(LEGACY_STORAGE_KEY)||'null';
    return normalizeLoadedData(JSON.parse(raw));
  }catch(e){
    return null;
  }
}
function saveToLocalStorage(data){
  try{
    const json=JSON.stringify(data);
    localStorage.setItem(STORAGE_KEY, json);
    localStorage.setItem(LEGACY_STORAGE_KEY, json);
    return true;
  }catch(e){
    return false;
  }
}
function openDB(){
  return new Promise((resolve,reject)=>{
    if(!('indexedDB' in window)) return reject(new Error('IndexedDB not supported'));
    const req=indexedDB.open(DB_NAME,1);
    req.onupgradeneeded=()=>{
      const db=req.result;
      if(!db.objectStoreNames.contains(STORE_NAME)) db.createObjectStore(STORE_NAME);
    };
    req.onsuccess=()=>resolve(req.result);
    req.onerror=()=>reject(req.error||new Error('DB open failed'));
  });
}
function loadFromIndexedDB(){
  return new Promise(async resolve=>{
    try{
      const db=await openDB();
      const tx=db.transaction(STORE_NAME,'readonly');
      const req=tx.objectStore(STORE_NAME).get(STORAGE_KEY);
      req.onsuccess=()=>resolve(normalizeLoadedData(req.result)||null);
      req.onerror=()=>resolve(null);
      tx.oncomplete=()=>db.close();
      tx.onerror=()=>db.close();
    }catch(e){
      resolve(null);
    }
  });
}
function saveToIndexedDB(data){
  return new Promise(async resolve=>{
    try{
      const db=await openDB();
      const tx=db.transaction(STORE_NAME,'readwrite');
      tx.objectStore(STORE_NAME).put(deepCopy(data), STORAGE_KEY);
      tx.oncomplete=()=>{ db.close(); resolve(true); };
      tx.onerror=()=>{ db.close(); resolve(false); };
    }catch(e){
      resolve(false);
    }
  });
}

  saveToLocalStorage(S.data);
}

}

function rowDateKey(r){
  if(r['日付']) return String(r['日付']).slice(0,10);
  return `${r['年']||0}-${String(r['月']||'').padStart(2,'0')}-${String(r['日']||'').padStart(2,'0')}`;
}
function sortByDateDesc(arr){
  return [...arr].sort((a,b)=>{
    const da=rowDateKey(a), db=rowDateKey(b);
    return da>db?-1:da<db?1:0;
  });
}

function buildSimpleForm(row,cfg){
  let html='';
  cfg.cols.forEach(c=>{
    const v=row[c]??'';
    if(c==='内容'||c==='詳細'||c==='ソース'){
      html+=`<div class="form-group"><label>${c}</label><textarea data-field="${c}">${esc(v)}</textarea></div>`;
    }else if(c==='区分'){
      const opts=DISTRICT_OPTIONS;
      html+=`<div class="form-group"><label>${c}</label><select data-field="${c}"><option value="">選択…</option>${opts.map(o=>`<option ${o===v?'selected':''}>${o}</option>`).join('')}</select></div>`;
    }else if(c==='仕入先'){
      const opts=[...SOURCE_OPTIONS,...getOptions('仕入先').filter(o=>!SOURCE_OPTIONS.includes(o))];
      html+=`<div class="form-group"><label>${c}</label><select data-field="${c}"><option value="">選択…</option>${opts.map(o=>`<option ${o===v?'selected':''}>${o}</option>`).join('')}</select></div>`;
    }else if(c==='取引先'&&cfg.combined){
      const opts=BUYER_OPTIONS;
      html+=`<div class="form-group"><label>販売先</label><select data-field="${c}"><option value="">選択…</option>${opts.map(o=>`<option ${o===v?'selected':''}>${o}</option>`).join('')}</select></div>`;
    }else if(c==='勘定科目'||c==='状況'){
      const opts=getOptions(c);
      html+=`<div class="form-group"><label>${c}</label><select data-field="${c}"><option value="">選択…</option>${opts.map(o=>`<option ${o===v?'selected':''}>${o}</option>`).join('')}</select></div>`;
    }else if(['年','月','日'].includes(c)){
      const sz=c==='年'?'4':c==='月'?'2':'2';
      html+=`<div class="form-group"><label>${c}</label><input type="text" inputmode="numeric" data-field="${c}" value="${esc(v)}" placeholder="${c}" maxlength="${sz}"></div>`;
    }else{
      const isNum=c.includes('金額')||c.includes('手数料')||c.includes('送料')||c.includes('利益')||c.includes('源泉');
      html+=`<div class="form-group"><label>${c}</label><input type="${isNum?'number':'text'}" inputmode="${isNum?'decimal':'text'}" data-field="${c}" value="${esc(v)}" step="${isNum?'1':''}" placeholder="${c}"></div>`;
    }
  });
  document.getElementById('entry-form').innerHTML=html;
  groupDateFields();
}

function getOptions(field){
  const sets={};
  ['売上帳','経費帳','仕入れ','源泉徴収管理','入金予定','帳簿外'].forEach(t=>{(S.data[t]||[]).forEach(r=>{if(r[field])sets[r[field]]=1;});});
  return Object.keys(sets).sort();
}

function closeModal(){document.getElementById('modal-overlay').classList.add('hidden');S.editing=null;}

function saveEntry(){
  const cfg=TABS[S.tab];
  const fields=document.querySelectorAll('#entry-form [data-field]');
  const row={};
  fields.forEach(f=>{
    const k=f.dataset.field;
    if(!k.startsWith('_')) row[k]=f.value.trim();
  });
  // normalize numbers
  cfg.cols.forEach(c=>{if(c.includes('金額')||c.includes('手数料')||c.includes('送料')||c.includes('利益')||c.includes('源泉'))row[c]=num(row[c]);});
  // Get expense fields (combined mode)
  const cost=num(document.getElementById('f-cost')?.value||0);
  const ship=num(document.getElementById('f-ship')?.value||0);
  const fee=num(document.getElementById('f-fee')?.value||0);
  if(S.editing!==null&&S.editing!==undefined){
    S.data[S.tab][S.editing]=row;
  }else{
    S.data[S.tab]=S.data[S.tab]||[];
    S.data[S.tab].push(row);
    // Auto-create expense entries for combined sales entry
    if(cfg.combined && (cost||ship||fee)){
      const y=row['年'],m=row['月'],d=row['日'],desc=row['内容']||'';
      S.data['経費帳']=S.data['経費帳']||[];
      if(cost) S.data['経費帳'].push({年:y,月:m,日:d,内容:'仕入 '+desc,勘定科目:'仕入れ',金額:cost,ソース:'売上入力より自動'});
      if(ship) S.data['経費帳'].push({年:y,月:m,日:d,内容:'送料 '+desc,勘定科目:'荷造運賃',金額:ship,ソース:'売上入力より自動'});
      if(fee) S.data['経費帳'].push({年:y,月:m,日:d,内容:'販売手数料 '+desc,勘定科目:'支払手数料',金額:fee,ソース:'売上入力より自動'});
    }
  }
  save();closeModal();render();
}

function deleteEntry(){
  if(S.editing!==null&&S.editing!==undefined){S.data[S.tab].splice(S.editing,1);save();closeModal();render();}
}

function exportSalesCSV(){
  const sales=S.data['売上帳']||[];
  if(!sales.length){alert('売上データがありません');return;}
  // Sort by date (年月日)
  const sorted=[...sales].sort((a,b)=>{
    const ka=`${a['年']||0}-${String(a['月']||'').padStart(2,'0')}-${String(a['日']||'').padStart(2,'0')}`;
    const kb=`${b['年']||0}-${String(b['月']||'').padStart(2,'0')}-${String(b['日']||'').padStart(2,'0')}`;
    return ka<kb?-1:ka>kb?1:0;
  });
  const headers=['年','月','日','販売先','内容','区分','仕入先','売上金額','ソース'];
  const lines=[headers.join(',')];
  sorted.forEach(r=>{
    const row=[r['年']||'',r['月']||'',r['日']||'',r['取引先']||'',r['内容']||'',r['区分']||'',r['仕入先']||'',r['金額']||0,r['ソース']||''];
    lines.push(row.map(v=>`"${String(v).replace(/"/g,'""')}"`).join(','));
  });
  // Add summary row
  const total=sorted.reduce((a,r)=>a+num(r['金額']),0);
  lines.push('');
  lines.push(`"合計","","","","","","","${total}",""`);
  const blob=new Blob(['\uFEFF'+lines.join('\n')],{type:'text/csv;charset=utf-8'});
  const a=document.createElement('a');a.href=URL.createObjectURL(blob);a.download=`売上データ_${new Date().toISOString().slice(0,10)}.csv`;a.click();
}

function exportExpenseCSV(){
  const exp=S.data['経費帳']||[];
  if(!exp.length){alert('経費データがありません');return;}
  // Sort by date (年月日)
  const sorted=[...exp].sort((a,b)=>{
    const ka=`${a['年']||0}-${String(a['月']||'').padStart(2,'0')}-${String(a['日']||'').padStart(2,'0')}`;
    const kb=`${b['年']||0}-${String(b['月']||'').padStart(2,'0')}-${String(b['日']||'').padStart(2,'0')}`;
    return ka<kb?-1:ka>kb?1:0;
  });
  const headers=['年','月','日','内容','勘定科目','金額','ソース'];
  const lines=[headers.join(',')];
  sorted.forEach(r=>{
    const row=[r['年']||'',r['月']||'',r['日']||'',r['内容']||'',r['勘定科目']||'',r['金額']||0,r['ソース']||''];
    lines.push(row.map(v=>`"${String(v).replace(/"/g,'""')}"`).join(','));
  });
  // Summary row
  const total=sorted.reduce((a,r)=>a+num(r['金額']),0);
  // Subtotal by 勘定科目
  const cats={};
  sorted.forEach(r=>{const c=r['勘定科目']||'その他';cats[c]=(cats[c]||0)+num(r['金額']);});

  lines.push(`"合計","","","","","${total}",""`);
  lines.push('');
  lines.push('"--- 勘定科目別内訳 ---"');
  Object.entries(cats).sort((a,b)=>b[1]-a[1]).forEach(([k,v])=>{
    lines.push(`"${k}","","","","","${v}",""`);
  });
  const blob=new Blob(['\uFEFF'+lines.join('\n')],{type:'text/csv;charset=utf-8'});

function restoreData(e){
  const file=e.target.files[0];
  if(!file)return;
  const reader=new FileReader();
  reader.onload=ev=>{
    try{
      const obj=JSON.parse(ev.target.result);
      if(!obj.data||typeof obj.data!=='object')throw new Error('invalid format');
      if(!confirm('現在のデータを上書きして復元しますか？\n（現在のデータは失われます）'))return;
      S.data=obj.data;
      save();render();
      alert('復元しました');
    }catch(err){
      alert('復元に失敗しました：バックアップファイルが正しくありません\n'+err.message);
    }
    e.target.value=''; // reset for next time
  };
  reader.readAsText(file,'utf-8');
}

function printAnnualReport(){

  const sales=(S.data['売上帳']||[]).filter(r=>String(r['年'])===String(year));

  const pur=(S.data['仕入れ']||[]).filter(r=>String(r['年'])===String(year));
  const wht=(S.data['源泉徴収管理']||[]).filter(r=>String(r['日付']||'').startsWith(String(year)));
  const off=(S.data['帳簿外']||[]).filter(r=>String(r['日付']||'').startsWith(String(year)));
